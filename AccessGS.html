/** * Access.gs: personalised, 24-hour access codes for the leadership assessments. * AccessCodes tab: Hash | Name | Email | Phone | Allowed | Created | Expires | CreatedBy | Status * Do not delete rows from the AccessCodes tab (switch a code off instead). */ // ---------- SETTINGS ---------- var ASSESS_URL = 'https://yourdomain.com/assess.html'; var ACCESS_HOURS = 24; var ACCESS_ASSESSMENTS = ['disc', 'eq', 'strengths', 'johari']; var ACCESS_ALPHABET = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // no 0/O or 1/I to avoid mix-ups // Which calls need a code, and for which assessment. Everything else keeps its own protection // (reading a result still needs the result ID plus the email; Johari peers use their private link). var GATED_FNS = { getQuestions: 'disc', saveResult: 'disc', johariCreateSession: 'johari', strengthsGetStatements: 'strengths', strengthsSaveResult: 'strengths', eqGetStatements: 'eq', eqSaveResult: 'eq' }; var ACCESS_NAMES = {disc: 'DISC Leadership Profile', eq: 'Emotional Intelligence', strengths: 'Strengths Compass', johari: 'Johari Window'}; function accessSheet_(ss) { var sh = ss.getSheetByName('AccessCodes') || ss.insertSheet('AccessCodes'); if (sh.getLastRow() === 0) { sh.appendRow(['Hash', 'Name', 'Email', 'Phone', 'Allowed', 'Created', 'Expires', 'CreatedBy', 'Status']); sh.setFrozenRows(1); } return sh; } function hashCode_(code) { var props = PropertiesService.getScriptProperties(); var pepper = props.getProperty('CODE_PEPPER'); if (!pepper) { pepper = Utilities.getUuid(); props.setProperty('CODE_PEPPER', pepper); } var norm = String(code || '').toUpperCase().replace(/[^A-Z0-9]/g, ''); return Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, pepper + '|' + norm)); } function newAccessCode_() { var hex = Utilities.getUuid().replace(/-/g, '').slice(0, 12) + Utilities.getUuid().replace(/-/g, '').slice(0, 12); var out = ''; for (var i = 0; i < 8; i++) out += ACCESS_ALPHABET.charAt(parseInt(hex.substr(i * 2, 2), 16) % 32); return 'NLI-' + out.slice(0, 4) + '-' + out.slice(4); } // Entry point for the assessment pages function assessRoute_(req) { try { if (req.fn === 'accessRedeem') return jsonOut({ok: true, data: accessRedeem_(req.code)}); var need = GATED_FNS[req.fn]; if (need) { var a = accessCheck_(req.token, need); if (req.payload) { req.payload.name = a.name; req.payload.email = a.email; } // identity comes from the code } } catch (err) { return jsonOut({ok: false, access: true, error: (err && err.message) || String(err)}); } return route_(req); } function accessRedeem_(code) { var cache = CacheService.getScriptCache(), tries = Number(cache.get('ac_tries') || 0); if (tries >= 30) throw new Error('Too many attempts. Please try again in 10 minutes.'); var h = hashCode_(code); var v = accessSheet_(SpreadsheetApp.getActiveSpreadsheet()).getDataRange().getValues(); for (var i = 1; i < v.length; i++) { if (v[i][0] !== h) continue; if (v[i][8] !== 'Active') throw new Error('This access code has been switched off.'); var exp = new Date(v[i][6]), left = Math.floor((exp - new Date()) / 1000); if (left <= 0) throw new Error('This access code has expired. Ask the school for a new one.'); var token = Utilities.getUuid() + Utilities.getUuid(), allowed = String(v[i][4]).split(','); var rec = {row: i + 1, name: v[i][1], email: v[i][2], allowed: allowed, expires: exp.toISOString()}; cache.put('acc_' + token, JSON.stringify(rec), Math.min(21600, left)); // sessions last up to 6 hours return {token: token, name: rec.name, allowed: allowed, expires: rec.expires}; } cache.put('ac_tries', tries + 1, 600); throw new Error('That code is not valid. Check it and try again.'); } function accessCheck_(token, need) { var raw = token ? CacheService.getScriptCache().get('acc_' + token) : null; if (!raw) throw new Error('Your access has ended. Please enter your access code again.'); var a = JSON.parse(raw); if (new Date() > new Date(a.expires)) throw new Error('This access code has expired. Ask the school for a new one.'); if (a.allowed.indexOf(need) < 0) throw new Error('Your code does not include this assessment.'); var status = accessSheet_(SpreadsheetApp.getActiveSpreadsheet()).getRange(a.row, 9).getValue(); if (status !== 'Active') throw new Error('This access code has been switched off.'); return a; } // Staff actions: createAccess, listAccess, revokeAccess (called from adminApi) function adminAccess_(d, ss, who) { if (['createAccess', 'listAccess', 'revokeAccess'].indexOf(d.action) < 0) return null; var sh = accessSheet_(ss), tz = Session.getScriptTimeZone(); if (d.action === 'createAccess') { var name = String(d.name || '').trim().slice(0, 80); var mail = String(d.clientEmail || '').trim().toLowerCase().slice(0, 120); var phone = String(d.phone || '').replace(/[^\d+]/g, '').slice(0, 20); var allowed = (d.allowed || []).filter(function (a) { return ACCESS_ASSESSMENTS.indexOf(a) >= 0; }); if (!name || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(mail) || !allowed.length) return jsonOut({ok: false, error: 'Enter the client name, a valid email and at least one assessment.'}); var code = newAccessCode_(), now = new Date(), exp = new Date(now.getTime() + ACCESS_HOURS * 3600000); sh.appendRow([hashCode_(code), safeCell(name), mail, phone, allowed.join(','), now, exp, who, 'Active']); var emailed = false; if (d.sendMail) { MailApp.sendEmail(mail, 'Your ' + SCHOOL_NAME + ' assessment access code', 'Hello ' + name + ',\n\nYour access code is: ' + code + '\n\nIt includes: ' + allowed.map(function (a) { return ACCESS_NAMES[a]; }).join(', ') + '\nIt is valid for ' + ACCESS_HOURS + ' hours (until ' + Utilities.formatDate(exp, tz, 'EEEE d MMMM, h:mm a') + ').\n\nOpen ' + ASSESS_URL + ' and enter the code to begin.\n\nThis code is personal to you. Please do not share it.'); emailed = true; } logChange(ss, who, 'createAccess', mail, allowed.join(','), '', ''); return jsonOut({ok: true, code: code, name: name, phone: phone, expires: exp.toISOString(), url: ASSESS_URL, emailed: emailed}); } if (d.action === 'listAccess') { var v = sh.getDataRange().getValues(), items = [], now2 = new Date(); for (var i = v.length - 1; i >= 1 && items.length < 50; i--) { var st = v[i][8] === 'Revoked' ? 'Revoked' : (now2 > new Date(v[i][6]) ? 'Expired' : 'Active'); items.push({row: i + 1, name: v[i][1], email: v[i][2], allowed: v[i][4], expires: new Date(v[i][6]).toISOString(), status: st}); } return jsonOut({ok: true, items: items}); } var rw = Number(d.row); // revokeAccess if (!(rw > 1 && rw <= sh.getLastRow())) return jsonOut({ok: false, error: 'Code not found.'}); sh.getRange(rw, 9).setValue('Revoked'); logChange(ss, who, 'revokeAccess', sh.getRange(rw, 3).getValue(), '', '', ''); return jsonOut({ok: true}); } var acc = adminAccess_(d, ss, who); if (acc) return acc;
